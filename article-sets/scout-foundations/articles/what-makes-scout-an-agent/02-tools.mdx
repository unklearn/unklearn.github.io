The runtime now passes tool schemas to the provider. It still cannot execute a request.

export function ApproveReader() {
  const s = useScoutStep(), ui = useLessonPresentation();
  const [approved, setApproved] = useState(false);
  return <>
    <p>Offering a reading tool is a configuration change. The runtime above stays the same when you approve it.</p>
    <RuntimeChange from={blocks.empty} to={blocks.reader} />
    <Action className="primary" disabled={approved} onClick={() => {
      s.setTools(blocks.reader.module.tools); s.runtime(blocks.offer);
      s.clearQuestion(); s.mark('approved'); setApproved(true);
      ui.renderSystemTerminal(<p>Reading tool available. No handler has executed.</p>);
      ui.showContext(s.schemas);
    }}>Approve</Action>
    {approved && <p>Continue to let Scout request this tool and execute it.</p>}
    <Action onClick={() => { s.reset({ tools: [] }); setApproved(false); }}>Reset</Action>
  </>;
}

<ApproveReader />

import type { RuntimeOptions } from '../../../../runtime/lesson-types.ts';

export class AgentRuntime {
  constructor(public options: RuntimeOptions) {}

  async evaluate() {
    const { provider, messages, tools, signal } = this.options;
    const schemas = tools.map(tool => tool.definition);
    return provider(messages, schemas, signal);
  }
}
