export function ModelTurn() {
  const s = useScoutStep(),
    [done, setDone] = useState(false);
  function run() {
    if (!s.restoreCheckpoint('posted-board')) s.reset({ tools: [] });
    if (!s.world.snapshot().notice) s.world.postNotice();
    s.setTools([]);
    return s.act(async () => {
      await s.runtime(blocks.invoke).evaluate();
      s.showContext({ outgoingMessages: s.messages, tools: s.schemas });
      s.showQuestion(
        'The reading capability is absent. No handler read the notice.',
      );
      s.saveCheckpoint('model-ready');
      s.mark('needs-tool');
      setDone(true);
    });
  }
  return (
    <>
      <Action className="primary" onClick={run}>
        Let Scout think
      </Action>
      {done && (
        <p>
          The terminal contains the actual public response. Whatever it
          proposes, no reading handler ran. Inspect Context: the exact order and
          stock were never sent. The question cloud marks the missing
          capability, not private model reasoning.
        </p>
      )}
    </>
  );
}
