export function Capabilities() {
  const s=useScoutStep(), ui=useLessonPresentation(), runtime=useRef(null), [phase,setPhase]=useState('ready');
  async function begin() {
    if(!s.restoreCheckpoint('observed-notice'))s.reset({tools:['read_notice']});s.setTools(['read_notice']);
    await s.act(async()=>{runtime.current=s.runtime(blocks.dynamic);await runtime.current.run();s.showQuestion('Movement and pantry tools have not been exposed.');s.mark('limited');setPhase('limited');});
  }
  return <>
    <Action onClick={begin}>Try the current tools</Action>
    {phase!=='ready' && <><p>Now change the tool configuration. This does not invoke the model, execute a tool, or construct another runtime.</p><RuntimeChange from={reading.reader} to={blocks.picnic} />
    <Action disabled={phase!=='limited'} onClick={()=>{s.setTools(blocks.picnic.module.tools);s.clearQuestion();s.showContext(s.schemas);setPhase('enabled');ui.renderSystemTerminal(<p>Delivery tools are available for the next turn. The runtime instance is unchanged.</p>);}}>Enable delivery tools</Action></>}
    {phase==='enabled' && <Action className="primary" onClick={()=>s.act(async()=>{
      const feedback=await runtime.current.execute(await runtime.current.evaluate());if(feedback)runtime.current.append(feedback);
      s.showContext({available:s.schemas,observation:feedback?.result});s.saveCheckpoint('expanded');s.mark('expanded');setPhase('acted');
    })}>Run the next turn</Action>}
    {phase==='acted' && <p>The terminal and world show the action actually requested and executed. In Mock, Scout walks to the pantry. Its next choice will still require an inspection; granting a tool did not reveal the stock.</p>}
    <p>Tool availability is configuration. The model still chooses whether to read, where to walk, when to inspect, and what to carry.</p>
  </>;
}

import type { Feedback, RuntimeOptions } from '../../../../runtime/lesson-types.ts';
import type { Tool } from '../../../../runtime/types.ts';
import type { Message } from '../../../../runtime/types.ts';

export class AgentRuntime {
  private turnTools: readonly Tool[] = [];
  constructor(public options: RuntimeOptions) {}

  async evaluate() {
    const { provider, messages, signal, getTools } = this.options;
    this.turnTools = getTools ? getTools() : this.options.tools;
    const schemas = this.turnTools.map(tool => tool.definition);
    return provider(messages, schemas, signal);
  }

  async execute(request: Message): Promise<Feedback | null> {
    const calls = request.tool_calls ?? [];
    if (calls.length > 1) throw new Error('Choose one tool per turn.');
    if (!calls.length) {
      if (!request.content.trim()) throw new Error('Empty provider response');
      return null;
    }
    const call = calls[0].function;
    const { signal, emit } = this.options;
    const tool = this.turnTools.find(tool => tool.definition.function.name === call.name);
    emit({ kind: 'tool_start', text: call.name, tool: call });
    const value = tool ? await tool.execute(call.arguments, signal)
      : { ok: false, reason: 'tool_unavailable' };
    signal.throwIfAborted();
    const result: Message = { role: 'tool', tool_name: call.name, content: JSON.stringify(value) };
    emit({ kind: 'tool_result', text: result.content, tool: call, result: value });
    return { request, result, value };
  }

  append({ request, result }: Feedback) {
    const { messages, emit } = this.options;
    messages.push(request);
    messages.push(result);
    emit({ kind: 'context', text: 'Request and observation appended', messages });
  }

  async run() {
    for (let turn = 0; turn < 20; turn++) {
      const request = await this.evaluate();
      const feedback = await this.execute(request);
      if (!feedback) {
        this.options.messages.push(request);
        return 'reported';
      }
      this.append(feedback);
    }
    return 'turn_limit';
  }
}
