export function Picnic() {
  const s = useScoutStep(), paused = useRef(false);
  const [phase, setPhase] = useState('ready');
  function run(fresh) {
    if (fresh) { s.reset({ tools: blocks.picnic.module.tools }); paused.current = false; }
    setPhase('running');
    return s.act(async () => {
      const runtime = s.runtime(blocks.loop);
      const status = await runtime.run(async feedback => {
        if (feedback.result.tool_name === 'inspect_pantry') {
          s.showContext({ observationInModelContext: feedback.result, messages: s.messages });
          if (s.settings.pauseInspection && !paused.current && feedback.value.ok) {
            paused.current = true; setPhase('inspected'); s.mark('inspected'); return false;
          }
        }
      });
      if (status !== 'paused') { setPhase('done'); s.mark(status); }
    });
  }
  return <>
    <p>Run with two apples. Then change only the starting apple count to one in Settings and run again. The same order produces an apple, then a carrot, because Scout learns the stock through inspection.</p>
    <div className="actions"><Action className="primary" onClick={() => run(true)}>Run picnic</Action><Action onClick={() => { s.reset({ tools: blocks.picnic.module.tools }); s.world.postNotice(); paused.current = false; setPhase('ready'); }}>Post a notice</Action></div>
    {phase === 'inspected' && <>
      <p>The observed stock is now in the next model input. Empty the apple shelf to make that observation stale, or continue unchanged.</p>
      <div className="actions"><Action onClick={() => { const stock = s.world.snapshot().pantry; s.world.setPantry({ ...stock, apple: 0 }); }}>Empty apple shelf</Action><Action className="primary" onClick={() => run(false)}>Continue</Action></div>
    </>}
    {phase === 'done' && <p>The basket and trace show the actual outcome. To compare orders, use two apples and post “Deliver one carrot to the picnic basket.” in Settings.</p>}
    <p>Taking the last apple is legal. “Leave at least one” belongs to the posted instruction, not the world rules.</p>
    <Action onClick={() => { s.reset({ tools: blocks.picnic.module.tools }); paused.current = false; setPhase('ready'); }}>Reset</Action>
  </>;
}

import type { Feedback, RuntimeOptions } from '../../../../runtime/lesson-types.ts';
import type { Message } from '../../../../runtime/types.ts';

export class AgentRuntime {
  constructor(public options: RuntimeOptions) {}

  async evaluate() {
    const { provider, messages, tools, signal } = this.options;
    const schemas = tools.map(tool => tool.definition);
    return provider(messages, schemas, signal);
  }

  async execute(request: Message): Promise<Feedback | null> {
    const calls = request.tool_calls ?? [];
    if (calls.length > 1) throw new Error('Choose one tool per turn.');
    if (!calls.length) {
      if (!request.content.trim()) throw new Error('Empty provider response');
      return null;
    }
    const call = calls[0].function;
    const { tools, signal, emit } = this.options;
    const tool = tools.find(tool => tool.definition.function.name === call.name);
    emit({ kind: 'tool_start', text: call.name, tool: call });
    const value = tool ? await tool.execute(call.arguments, signal)
      : { ok: false, reason: 'tool_unavailable' };
    signal.throwIfAborted();
    const result: Message = { role: 'tool', tool_name: call.name, content: JSON.stringify(value) };
    emit({ kind: 'tool_result', text: result.content, tool: call, result: value });
    return { request, result, value };
  }

  append({ request, result }: Feedback) {
    const { messages, emit } = this.options;
    messages.push(request);
    messages.push(result);
    emit({ kind: 'context', text: 'Request and observation appended', messages });
  }

  async run(afterTool?: (feedback: Feedback) => Promise<boolean | void>) {
    for (let turn = 0; turn < 20; turn++) {
      const request = await this.evaluate();
      const feedback = await this.execute(request);
      if (!feedback) {
        this.options.messages.push(request);
        return 'reported';
      }
      this.append(feedback);
      if (await afterTool?.(feedback) === false) return 'paused';
    }
    return 'turn_limit';
  }
}
