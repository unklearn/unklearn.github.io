export function AutonomousPicnic() {
  const [prediction, setPrediction] = useState(null);
  const s = useScoutStep(),
    [apples, setApples] = useState(2);
  const runs = [s.recall('picnic-2'), s.recall('picnic-1')].filter(Boolean);
  const two = s.recall('picnic-2'),
    one = s.recall('picnic-1');
  const inspected = (run) =>
    run?.status === 'reported' &&
    run.events.some(
      (e) =>
        e.kind === 'tool_result' &&
        e.tool?.name === 'inspect_pantry' &&
        e.result?.ok,
    );
  const delivered = (run) =>
    run?.final?.basket.apple === 1 && run.final.basket.carrot === 0
      ? 'apple'
      : run?.final?.basket.carrot === 1 && run.final.basket.apple === 0
        ? 'carrot'
        : null;
  const comparison =
    inspected(two) &&
    inspected(one) &&
    two.initial.notice === one.initial.notice &&
    delivered(two) &&
    delivered(one)
      ? delivered(two) === delivered(one)
        ? 'same-item'
        : 'different-items'
      : null;
  return (
    <>
      <Prediction
        question={meta.question}
        choices={[
          {
            id: 'same-item',
            label: 'Scout will deliver the same item in both runs',
            explanation:
              'Both completed runs delivered the same item despite their different observed stock. These are the actual outcomes; a Live model may interpret the order differently. Matching choices do not prove that the route was scripted.',
          },
          {
            id: 'different-items',
            label: 'Scout will deliver different items after inspecting stock',
            explanation:
              'The same order produced different delivered items after different stock observations. Inspection results entered the conversation before pickup; the world did not enforce leaving the last apple.',
          },
        ]}
        value={prediction}
        onChange={setPrediction}
        outcome={!s.busy ? comparison : null}
      />
      <p>The posted order is:</p>
      <blockquote>{s.settings.order}</blockquote>
      <fieldset className="preset-choice">
        <legend>Starting pantry</legend>
        {[2, 1].map((n) => (
          <label key={n}>
            <input
              type="radio"
              name="picnic-apples"
              checked={apples === n}
              onChange={() => setApples(n)}
            />
            {n} apple{n === 1 ? '' : 's'}, 2 carrots
          </label>
        ))}
      </fieldset>
      <Action
        className="primary"
        onClick={() => blocks.autonomous.module.run(s, apples)}
      >
        Run picnic
      </Action>
      <p>
        Run with two apples. Then change only the starting apples to one and run
        again. Each run resets Scout’s world, hands, basket, and conversation.
        Settings is available for manually posting a different order.
      </p>
      <RuntimeChange to={blocks.autonomous} />
      {runs.length > 0 && (
        <>
          <h2 className="outcome-title">Observed outcomes</h2>
          <table className="outcomes">
            <thead>
              <tr>
                <th>Starting apples</th>
                <th>Basket</th>
                <th>Pantry left</th>
              </tr>
            </thead>
            <tbody>
              {runs.map((run) => (
                <tr key={run.id}>
                  <td>{run.initial.pantry.apple}</td>
                  <td>
                    {run.final.basket.apple} apple, {run.final.basket.carrot}{' '}
                    carrot
                  </td>
                  <td>
                    {run.final.pantry.apple} apple, {run.final.pantry.carrot}{' '}
                    carrot
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
          <p>
            These are actual inventories, not expected-answer fixtures. Open
            History to follow inspection, pickup, and delivery. Live can make a
            different choice; the world and trace preserve it.
          </p>
        </>
      )}
      <p>
        Mock learns stock only from <code>inspect_pantry</code>. With the
        default notice it takes an apple from two, and a carrot when only one
        apple remains. Taking the last apple is legal: preserving it is the
        model’s instruction, not a physical rule.
      </p>
    </>
  );
}

import type { ScoutStep } from '../../session.ts';
import { blocks as expanded } from './06-capabilities.mdx';
export async function run(s: ScoutStep, apples = 2) {
  s.reset({
    pantry: { apple: apples, carrot: 2 },
    tools: expanded.picnic.module.tools,
  });
  await s.act(async () => {
    const runtime = s.runtime(expanded.dynamic);
    s.mark(await runtime.run!()); // No scripted route, item choice, or intermediate pause.
  });
  if (s.trace.final && s.trace.status !== 'cancelled') {
    s.saveCheckpoint('picnic-complete');
    s.remember('picnic-latest', s.trace);
    s.remember(`picnic-${apples}`, s.trace);
  }
}
