export const meta = {
  id: 'meet',
  title: 'Following a script',
  question: 'Will posting a notice change a scripted patrol?',
  initial: { tools: [], notice: '' },
  narration: {
    text: 'Meet Scout, the courier. A patrol can move its body, but it does not interpret the job on a notice.',
  },
};

Scout is a courier. Its job is to carry one item from the pantry, across the bridge, to the picnic basket. We will build the runtime that lets it choose what to do.

export function ScriptedPatrol() {
  const [prediction, setPrediction] = useState(null);
  const s = useScoutStep(),
    [phase, setPhase] = useState('ready');
  return (
    <>
      <Prediction
        question={meta.question}
        choices={[
          {
            id: 'reads',
            label: 'Scout will read the notice',
            explanation:
              'Scout read the notice. The notice-read event, rather than movement alone, is evidence of reading.',
          },
          {
            id: 'patrol',
            label: 'Scout will follow its route',
            explanation:
              'Scout followed its authored path and left the posted notice unread. The patrol script never checked the board; posting a notice did not change its behavior.',
          },
        ]}
        value={prediction}
        onChange={setPrediction}
        outcome={
          !s.busy &&
          s.stage === 'passed' &&
          s.trace.events.some((e) => e.kind === 'notice_posted')
            ? s.world.snapshot().unread
              ? 'patrol'
              : 'reads'
            : null
        }
      />
      <p>
        {phase === 'walking'
          ? 'Post a picnic notice while Scout walks. The amber light means a notice is waiting.'
          : 'Start the patrol'}
      </p>
      <div className="actions">
        <Action
          className="primary"
          onClick={async () => {
            setPhase('walking');
            await blocks.patrol.module.run(s);
            if (s.stage === 'passed') setPhase('passed');
          }}
        >
          Start patrol
        </Action>
        <Action
          concurrent
          disabled={phase === 'ready'}
          onClick={() => s.world.postNotice()}
        >
          Post a notice
        </Action>
      </div>
      {phase === 'passed' && (
        <p>
          Scout finished its authored path.{' '}
          {s.world.snapshot().unread
            ? 'The notice is still unread.'
            : 'Post a notice, then continue.'}{' '}
          Movement alone did not interpret a job. This experiment made no model
          calls and executed no tools.
        </p>
      )}
      <RuntimeChange to={blocks.patrol} />
    </>
  );
}

<ScriptedPatrol />

    s.saveCheckpoint('posted-board');

}
