The AI workflow let a model choose an item inside a fixed sequence. Now we will let it choose what to do next. Start with a fresh conversation: Scout receives only a new-notice notification, without the notice text, stock, or tools. A thinking bubble marks an actual pending provider call.

export function ModelTurn() {
  const [prediction, setPrediction] = useState(null);
  const s = useScoutStep(),
    [done, setDone] = useState(false),
    [response, setResponse] = useState(null);
  function run() {
    setResponse(null);
    setDone(false);
    s.reset({ notice: '', tools: [] });
    s.world.postNotice();
    s.setTools([]);
    return s.act(async () => {
      setResponse(await s.runtime(blocks.invoke).evaluate());
      s.showContext({ outgoingMessages: s.messages, tools: s.schemas });
      s.showQuestion(
        'The reading capability is absent. No handler read the notice.',
      );
      s.saveCheckpoint('model-ready');
      s.mark('needs-tool');
      setDone(true);
    });
  }
  return (
    <>
      <Prediction
        question={meta.question}
        choices={[
          {
            id: 'world',
            label: 'The posted order and the pantry stock',
            explanation:
              'The input contained observations of the order and stock. Models can use information that is included in their conversation.',
          },
          {
            id: 'conversation',
            label: 'Only the conversation sent to it',
            explanation:
              'The outgoing input contained a new-notice notification, not the notice text or pantry stock. Running the model did not supply the missing observations.',
          },
        ]}
        value={prediction}
        onChange={setPrediction}
        outcome={!s.busy && s.stage === 'needs-tool' ? 'conversation' : null}
      />
      <RuntimeChange to={blocks.invoke} />
      <Action className="primary" onClick={run}>
        Let Scout think
      </Action>
      {done && response?.content && (
        <blockquote className="public-response">{response.content}</blockquote>
      )}
      {done && (
        <p>
          This is the provider’s actual public response. Whatever it proposes,
          no reading handler ran. Context in System terminal shows the exact
          order and stock were never sent. The question cloud marks the missing
          capability, not private model reasoning.
        </p>
      )}
    </>
  );
}

export class AgentRuntime {
  constructor(public options: RuntimeOptions) {}

  async evaluate() {
    const { provider, messages, signal } = this.options;
    return provider(messages, [], signal);
  }
}
